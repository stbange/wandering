cmake_minimum_required(VERSION 3.20)
project(WanderingCastle VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(WC_COVERAGE "Instrument the authoritative simulation" OFF)
option(WC_BUILD_GAME "Build SDL3/OpenGL application" ON)
option(WC_FETCH_DEPS "Download pinned dependencies when absent" ON)
include(FetchContent)
add_library(wc_core src/core/world.c src/core/inventory.c src/core/simulation.c src/core/commands.c src/core/save.c)
target_include_directories(wc_core PUBLIC src)
if(NOT MSVC)
  target_compile_options(wc_core PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(wc_core PUBLIC m)
endif()
if(WC_COVERAGE AND NOT EMSCRIPTEN)
  target_compile_options(wc_core PRIVATE --coverage -O0 -g)
  target_link_options(wc_core INTERFACE --coverage)
endif()

if(EXISTS "${CMAKE_SOURCE_DIR}/.deps/lua-5.4.8/src/lapi.c")
  set(lua_SOURCE_DIR "${CMAKE_SOURCE_DIR}/.deps/lua-5.4.8")
else()
  FetchContent_Declare(lua URL https://www.lua.org/ftp/lua-5.4.8.tar.gz URL_HASH SHA256=4f18ddae154e793e46eeab727c59ef1c0c0c2b744e7b94219710d76f530629ae)
  FetchContent_MakeAvailable(lua)
endif()
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS "${lua_SOURCE_DIR}/src/*.c")
list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|luac)\\.c$")
add_library(wc_lua STATIC ${LUA_SOURCES})
target_include_directories(wc_lua PUBLIC "${lua_SOURCE_DIR}/src")
add_library(wc_content src/content.c)
target_link_libraries(wc_content PUBLIC wc_core wc_lua)

enable_testing()
if(NOT EMSCRIPTEN)
  add_executable(wc_tests tests/test_core.c)
  target_link_libraries(wc_tests PRIVATE wc_content)
  add_test(NAME core COMMAND wc_tests "${CMAKE_SOURCE_DIR}/assets/scripts/content.lua")
endif()
if(WC_BUILD_GAME)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_RENDER OFF CACHE BOOL "" FORCE)
  set(SDL_GPU OFF CACHE BOOL "" FORCE)
  set(SDL_VULKAN OFF CACHE BOOL "" FORCE)
  set(SDL_CAMERA OFF CACHE BOOL "" FORCE)
  if(EXISTS "${CMAKE_SOURCE_DIR}/.deps/SDL3-3.2.16/CMakeLists.txt")
    add_subdirectory(.deps/SDL3-3.2.16 EXCLUDE_FROM_ALL)
  else()
    FetchContent_Declare(SDL3 URL https://github.com/libsdl-org/SDL/releases/download/release-3.2.16/SDL3-3.2.16.tar.gz URL_HASH SHA256=6340e58879b2d15830c8460d2f589a385c444d1faa2a4828a9626c7322562be8)
    FetchContent_MakeAvailable(SDL3)
  endif()
  add_executable(wandering src/main.c src/render.c src/platform.c)
  target_link_libraries(wandering PRIVATE wc_content SDL3::SDL3)
  target_compile_definitions(wandering PRIVATE WC_BUILD_VERSION="${PROJECT_VERSION}")
  if(EMSCRIPTEN)
    set_target_properties(wandering PROPERTIES SUFFIX ".html")
    target_compile_options(wandering PRIVATE -O2)
    target_link_options(wandering PRIVATE -O2 -sUSE_WEBGL2=1 -sMIN_WEBGL_VERSION=2 -sMAX_WEBGL_VERSION=2 -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=2097152 -sFORCE_FILESYSTEM=1 -sEXPORTED_RUNTIME_METHODS=ccall,FS -sEXPORTED_FUNCTIONS=_main,_wc_web_action,_wc_web_look,_wc_web_move,_wc_web_snapshot,_wc_xr_view,_wc_xr_frame,_wc_xr_end,_malloc,_free "-sDEFAULT_LIBRARY_FUNCS_TO_INCLUDE=$WanderingXR" -lidbfs.js --shell-file "${CMAKE_SOURCE_DIR}/web/shell.html" --js-library "${CMAKE_SOURCE_DIR}/web/xr.js")
  else()
    if(APPLE)
      target_link_libraries(wandering PRIVATE "-framework OpenGL")
    elseif(WIN32)
      target_link_libraries(wandering PRIVATE opengl32)
    else()
      find_library(WC_GL_LIBRARY NAMES GL OpenGL libGL.so.1 REQUIRED)
      target_link_libraries(wandering PRIVATE ${WC_GL_LIBRARY})
    endif()
    install(TARGETS wandering RUNTIME DESTINATION .)
    install(DIRECTORY assets DESTINATION .)
    install(FILES LICENSE README.md DESTINATION .)
    install(DIRECTORY docs DESTINATION .)
    install(DIRECTORY openspec DESTINATION .)
  endif()
  add_custom_command(TARGET wandering POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:wandering>/assets")
  add_custom_command(TARGET wandering POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/LICENSE" "${CMAKE_SOURCE_DIR}/README.md" "$<TARGET_FILE_DIR:wandering>")
  add_custom_command(TARGET wandering POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/docs" "$<TARGET_FILE_DIR:wandering>/docs")
  add_custom_command(TARGET wandering POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/openspec" "$<TARGET_FILE_DIR:wandering>/openspec")
endif()
