cmake_minimum_required(VERSION 3.24)
project(wandering VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(WC_COVERAGE "Instrument authored core for gcov" OFF)
option(WC_BUILD_GAME "Build SDL3/OpenGL application" ON)
set(WC_LUA_SOURCE "" CACHE PATH "Path to extracted Lua 5.4.8 source directory")
include(FetchContent)
if(NOT WC_LUA_SOURCE)
  FetchContent_Declare(lua_source URL https://www.lua.org/ftp/lua-5.4.8.tar.gz
    URL_HASH SHA256=4f18ddae154e793e46eeab727c59ef1c0c0c2b744e7b94219710d76f530629ae)
  FetchContent_MakeAvailable(lua_source)
  set(WC_LUA_SOURCE "${lua_source_SOURCE_DIR}")
endif()
file(GLOB LUA_FILES CONFIGURE_DEPENDS "${WC_LUA_SOURCE}/src/*.c")
list(FILTER LUA_FILES EXCLUDE REGEX "/(lua|luac)\\.c$")
add_library(wc_lua STATIC ${LUA_FILES})
target_include_directories(wc_lua PUBLIC "${WC_LUA_SOURCE}/src")
if(UNIX)
  target_link_libraries(wc_lua PUBLIC m ${CMAKE_DL_LIBS})
endif()
add_library(wc_core src/core.c src/save.c src/content.c)
target_include_directories(wc_core PUBLIC src)
target_link_libraries(wc_core PUBLIC wc_lua)
if(MSVC)
  target_compile_options(wc_core PRIVATE /W4)
  target_compile_definitions(wc_core PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(wc_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(NOT WIN32)
    target_link_libraries(wc_core PUBLIC m)
  endif()
endif()
if(WC_COVERAGE)
  target_compile_options(wc_core PRIVATE --coverage -O0 -g)
  target_link_options(wc_core PUBLIC --coverage)
endif()
enable_testing()
add_executable(wc_tests tests/test_core.c)
target_link_libraries(wc_tests PRIVATE wc_core)
add_test(NAME core COMMAND wc_tests "${CMAKE_SOURCE_DIR}/content/home.lua")
set_tests_properties(core PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}")
if(WC_BUILD_GAME)
  find_package(SDL3 CONFIG QUIET)
  if(NOT SDL3_FOUND)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL3 URL https://github.com/libsdl-org/SDL/releases/download/release-3.4.20/SDL3-3.4.20.tar.gz)
    FetchContent_MakeAvailable(SDL3)
  endif()
  add_executable(wandering src/main.c src/render.c src/input.c)
  target_link_libraries(wandering PRIVATE wc_core SDL3::SDL3)
  if(EMSCRIPTEN)
    set_target_properties(wandering PROPERTIES SUFFIX .html)
    target_link_options(wandering PRIVATE -sUSE_WEBGL2=1 -sMIN_WEBGL_VERSION=2 -sMAX_WEBGL_VERSION=2
      -sALLOW_MEMORY_GROWTH=1 -sFORCE_FILESYSTEM=1 -lidbfs.js
      "--preload-file" "${CMAKE_SOURCE_DIR}/content@/content"
      "--preload-file" "${CMAKE_SOURCE_DIR}/assets@/assets"
      "--shell-file" "${CMAKE_SOURCE_DIR}/web/shell.html")
  else()
    find_package(OpenGL REQUIRED)
    target_link_libraries(wandering PRIVATE OpenGL::GL)
    if(WIN32)
      add_custom_command(TARGET wandering POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:SDL3::SDL3-shared> $<TARGET_FILE_DIR:wandering>)
    endif()
    add_test(NAME game COMMAND wandering --game-test)
  endif()
  foreach(dir content assets)
    add_custom_command(TARGET wandering POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
      "${CMAKE_SOURCE_DIR}/${dir}" "$<TARGET_FILE_DIR:wandering>/${dir}")
    install(DIRECTORY "${dir}" DESTINATION .)
  endforeach()
  install(TARGETS wandering RUNTIME DESTINATION .)
  if(WIN32)
    install(FILES $<TARGET_FILE:SDL3::SDL3-shared> DESTINATION .)
  endif()
endif()
install(FILES LICENSE README.md DESTINATION .)
install(DIRECTORY docs DESTINATION .)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "wandering-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}")
include(CPack)
